        // Navigation Controller
        function switchPage(pageId) {
            const pages = ['home', 'our-pets', 'adoption-process', 'contact'];
            pages.forEach(p => {
                const pageEl = document.getElementById(`page-${p}`);
                const navEl = document.getElementById(`nav-${p}`);
                if (pageEl) pageEl.classList.add('hidden');
                if (navEl) navEl.classList.remove('active');
            });

            const activePage = document.getElementById(`page-${pageId}`);
            const activeNav = document.getElementById(`nav-${pageId}`);
            if (activePage) activePage.classList.remove('hidden');
            if (activeNav) activeNav.classList.add('active');

            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        function toggleMobileMenu() {
            const menu = document.getElementById('mobile-menu');
            menu.classList.toggle('hidden');
        }

            const container = document.getElementById('pet-grid');

                        <div class="relative h-60 overflow-hidden">
                            <img src="${pet.image}" alt="${pet.name}" class="w-full h-full object-cover transform hover:scale-110 transition duration-500" onerror="this.src='https://placehold.co/600x400/EFE3D3/3E362E?text=${pet.name}'">
                            <span class="absolute top-4 right-4 bg-nude-100/90 backdrop-blur-md text-nude-800 text-xs font-bold px-3 py-1 rounded-full shadow border border-nude-400">
                                ${pet.category}
                            </span>
                        </div>
                        <div class="p-6 flex-grow flex flex-col justify-between space-y-4">
                            <div>

                                    <h3 class="text-2xl font-bold text-nude-800">${pet.name}</h3>

                                <p class="text-xs text-nude-600 font-medium mt-1">Breed: ${pet.breed} &bull; Age: ${pet.age}</p>

                                    Adopt Me
                                </button>
                            </div>
                        </div>
                    </div>
                `;
                container.innerHTML += cardHtml;
            });
        }

        // Filtering Logic
        function filterPets(category) {
            currentFilter = category;
            
            const filterBtns = document.querySelectorAll('.filter-btn');
            filterBtns.forEach(btn => {
                if (btn.innerText.toLowerCase().includes(category.toLowerCase()) || (category === 'all' && btn.innerText.includes('All'))) {
                    btn.className = 'filter-btn bg-nude-500 text-white font-medium px-5 py-2 rounded-full text-sm transition';
                } else {
                    btn.className = 'filter-btn bg-nude-100 hover:bg-nude-300 text-nude-800 font-medium px-5 py-2 rounded-full text-sm transition';
                }
            });

            if (category === 'all') {
                renderPetGrid(petData);
            } else {
                const filtered = petData.filter(p => p.category.toLowerCase() === category.toLowerCase());
                renderPetGrid(filtered);
            }
        }

        // "Adopt Me" Popup Alert Function
        function adoptAlert(petName) {
            window.alert("Thank you for your interest in adopting! Please visit our Contact page to fill out an application.");
            switchPage('contact');
            const selectEl = document.getElementById('contact-pet-select');
            if (selectEl) {
                for (let i = 0; i < selectEl.options.length; i++) {
                    if (selectEl.options[i].value.startsWith(petName)) {
                        selectEl.selectedIndex = i;
                        break;
                    }
                }
            }
        }

        // Contact Form Submission
        function handleFormSubmit(e) {
            e.preventDefault();
            const name = document.getElementById('contact-name').value;
            const pet = document.getElementById('contact-pet-select').value;
            
            showToast(`Thank you ${name}! Your application for ${pet || 'adoption'} was submitted.`);
            document.getElementById('adoption-contact-form').reset();
        }

        // Helper: Scroll to table
        function scrollToBaselineTable() {
            document.getElementById('baseline-table-section').scrollIntoView({ behavior: 'smooth' });
        }

        // Custom Toast Notification System
        function showToast(message) {
            const container = document.getElementById('toast-container');
            const toast = document.createElement('div');
            toast.className = 'toast-animate bg-nude-800 text-nude-100 text-xs sm:text-sm px-5 py-3 rounded-2xl shadow-xl flex items-center space-x-3 pointer-events-auto border border-nude-500/40';
            toast.innerHTML = `
                <i class="fa-solid fa-circle-check text-nude-500 text-base"></i>
                <span>${message}</span>
            `;
            container.appendChild(toast);
            setTimeout(() => {
                toast.style.opacity = '0';
                toast.style.transition = 'opacity 0.4s ease';
                setTimeout(() => toast.remove(), 400);
            }, 3000);
        }
